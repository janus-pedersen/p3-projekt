\subsection{Cekura}
smykke 
\begin{itemize}
    \item anti stigmatiserende
    \item kommer i kontakt med vagtcentral eller med pårørende
    \item virker i Danmark
    \item forkellige pakker:
\end{itemize}
    \begin{enumerate}
        \item begrænset - kun til pårørende
        \begin{enumerate}
            \item man får hovedprodukt
            \item evighedsgaranti
            \item servicedækning i hele landet
            \item 'aktiv sim-kort med fri trafik indbygget'
        \end{enumerate}
        \item standard - til vagtcentral
        \begin{enumerate}
            \item ovenstående + vagtcentral
        \end{enumerate}
        \item individuel - tryghed efter dine behov
        \begin{enumerate}
            \item ovenstående + teknisk overvågning + batterivarsling
            \item individuel tilpasning af servicen
            \item valgfrit om alarm skal skal besvares af central eller kontaktperson
            \item tryghedssamtaler ved ensomhed
            \item tryghedsopkald om alt er ok
        \end{enumerate}
    \end{enumerate}
\begin{itemize}
    \item markedsført mod ældre, bevægelseshæmmede, kroniske syge og psykisk sårbare
\end{itemize}
